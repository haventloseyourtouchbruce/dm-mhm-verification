cmake_minimum_required(VERSION 3.22)
project(dm_mhm_verification VERSION 2.0 LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# Compile checks only: no matcher implementation is built or linked.
add_library(header_c OBJECT tests/header_c.c)
add_library(header_cpp OBJECT tests/header_cpp.cpp)
foreach(target IN ITEMS header_c header_cpp)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(BUILD_TESTING)
  add_test(NAME runner_unit_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -v)
  add_test(NAME fixture_integrity COMMAND "${Python3_EXECUTABLE}" -m harness.grade)
  set_tests_properties(runner_unit_tests fixture_integrity PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
